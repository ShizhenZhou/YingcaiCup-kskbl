# ══════════════════════════════════════════════════════════════
# 一、系统依赖（apt 装，不是 pip）
# ══════════════════════════════════════════════════════════════
#   sudo apt install -y \
#        python3-libgpiod python3-spidev python3-pil python3-numpy \
#        python3-rknnlite2 libportaudio2 fonts-noto-cjk
#
#   python3-libgpiod  屏幕 DC/RES 的 GPIO 控制
#   python3-spidev    SPI（屏幕数据传输）
#   python3-pil       画字画图
#   python3-rknnlite2 NPU 推理（MeloTTS 解码器）—— **只能 apt 装**
#   libportaudio2     sounddevice 的底层
#   fonts-noto-cjk    屏幕显示中英文（别用 DroidSansFallback，它不含拉丁字母）

# ══════════════════════════════════════════════════════════════
# 二、venv 必须能看到系统包
# ══════════════════════════════════════════════════════════════
# rknnlite / spidev / gpiod / PIL 都是 apt 包，venv 默认看不到。
# 加一个 .pth 即可（等价于 python3 -m venv --system-site-packages）：
#
#   echo /usr/lib/python3/dist-packages \
#     > ~/proj/venv/lib/python3.11/site-packages/system-packages.pth
